End-to-End Example: H3 Grid + NDVI + Sophia AI

import std.io
import std.gis.geolibre
import std.gis.sophia_copilot

fn main() {
    // 1. Index Lagos & Abuja under Uber H3 Discrete Global Grid
    let lagos_h3 = geolibre.latlng_to_h3(6.5244, 3.3792, 8)
    let abuja_h3 = geolibre.latlng_to_h3(9.0765, 7.3986, 8)
    println("Lagos H3 Index: 0x" + lagos_h3.to_hex())

    // 2. Geodesic Distance
    let dist_m = geolibre.distance_haversine(6.5244, 3.3792, 9.0765, 7.3986)
    println("Lagos to Abuja Distance: " + (dist_m / 1000.0).to_string() + " km")

    // 3. Sophia Natural Language Spatial Reasoning
    let prompt = "Sophia, find high-canopy agricultural parcels and density clusters in FCT Abuja"
    let query_result = sophia_copilot.execute_intent(prompt)
    println("Sophia Reasoning: " + query_result.reasoning_summary)
    println("Compiled Spatial SQL: " + query_result.generated_spatial_sql)
    println("Deck.gl Visualizer Config: " + query_result.deckgl_json)
}
