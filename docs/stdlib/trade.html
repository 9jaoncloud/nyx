1. Compiling & Backtesting Pine Script Strategies

Transpiling Pine Script to SIMD Vector IR
import "projects/nyx-trade-engine/src/trade_engine.nyx";

fn main() {
    region trade_arena {
        let engine = NyxTradeEngine.new("BTC/USDT");

        // Execute Pine Script v5 Strategy over 1,000,000 historical candles
        let pine_code = "
            //@version=5
            strategy('EMA_Cross_RSI_Reversal', overlay=true)
            fast_ema = ta.ema(close, 9)
            slow_ema = ta.ema(close, 21)
            rsi = ta.rsi(close, 14)
            if (ta.crossover(fast_ema, slow_ema) and rsi < 30)
                strategy.entry('Long', strategy.long)
        ";

        let signal = engine.execute_pinescript_strategy(pine_code, 1_000_000);
        println("Signal Generated: ", signal.indicator_name, " | Confidence: ", signal.confidence_score);
    }
}
