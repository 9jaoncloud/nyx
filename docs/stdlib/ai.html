1. Running 700B+ MoE Models on Consumer Hardware

FreeToken Dynamic Layer Streaming
import "projects/nyx-ai-engine/src/ai_engine.nyx";

fn main() {
    region inference_arena {
        // Initialize 700B MoE model with 64 total experts, Top-8 active routing
        let engine = NyxAiEngine.new(
            model: "DeepSeek-V3-700B-MoE",
            params_b: 671.0,
            quant: "FP8",
            vram_mb: 16384 // 16GB VRAM limit
        );

        // Stream and infer transformer layer with dynamic DMA prefetch
        let tps = engine.stream_and_infer_layer(layer_index: 12, input_tokens: 2048);
        println("Generated throughput: ", tps, " tokens/sec");

        // Execute Sliding Window Attention for ultra-fast local context
        let swa_tps = engine.execute_swa_attention(window_size: 4096, seq_length: 32768);
        println("SWA throughput: ", swa_tps, " tokens/sec");
    }
}
