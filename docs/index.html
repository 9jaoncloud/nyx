⚡ std.web Reactive Microservice

import std.web;

fn main() {
    let app = WebApp.new();
    app.get("/events", fun(req) {
        return Response::sse(fun(stream) {
            stream.send("tick", now_iso8601());
        });
    });
    app.listen("0.0.0.0:8080");
}

📱 sdk.ui.mobile_sdk Component

import sdk.ui.mobile_sdk;

fn CounterWidget() -> Widget {
    let count = Signal.new(0);
    return Column([
        Text("Taps: ${count.value}"),
        Button("Increment", fun() {
            count.update(fun(c) { return c + 1; });
        })
    ]);
}

🤖 Tri-Mode Android CLI

# Inspect manifest and permissions
nyx apk inspect app.apk

# Transpile AAB bundle to target APK
nyx aab transpile bundle.aab --out=app.apk

# Execute via native NDK or sandbox
nyx apk run app.apk --mode=ndk
nyx apk run app.apk --mode=container
