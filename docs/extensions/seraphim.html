2. Spatial Force Backward Recursion & Joint Torques: $$\mathbf{f}_i = \mathbf{I}_i \mathbf{a}_i + \mathbf{v}_i \times^* (\mathbf{I}_i \mathbf{v}_i) + \sum_{j \in \mu(i)} \mathbf{X}_j^* \mathbf{f}_j$$ $$\tau_i = \mathbf{S}_i^T \mathbf{f}_i + b_i \dot{q}_i$$
3. Whole-Body Control (WBC) Quadratic Programming: $$\min_{\ddot{\mathbf{q}}, \mathbf{f}_c} \frac{1}{2} \|\mathbf{J}_{\text{task}} \ddot{\mathbf{q}} - \ddot{\mathbf{x}}_{\text{des}}\|_{\mathbf{W}_t}^2 + \frac{1}{2} \|\boldsymbol{\tau}\|_{\mathbf{W}_\tau}^2 + \frac{1}{2} \|\mathbf{f}_c\|_{\mathbf{W}_f}^2$$ $$\text{subject to: } \mathbf{M}(\mathbf{q})\ddot{\mathbf{q}} + \mathbf{h}(\mathbf{q}, \dot{\mathbf{q}}) = \mathbf{S}^T \boldsymbol{\tau} + \mathbf{J}_c^T \mathbf{f}_c, \quad \mathbf{f}_c \in \mathcal{K}_{\text{friction}}, \quad \boldsymbol{\tau}_{\min} \le \boldsymbol{\tau} \le \boldsymbol{\tau}_{\max}$$

Use Case 1: Autonomous 34-DoF Bipedal Locomotion & Stair Climbing

In dynamic stair ascent, a 34-DoF humanoid must continually recalculate its Capture Point (CP) and adjust its Zero Moment Point (ZMP) under friction cone constraints: 

import projects.seraphim.client;
import projects.seraphim.kinematics;
import std.hardware.ethercat;

pub fn run_stair_climbing_controller(model_path: String) {
    let mut humanoid = SeraphimHumanoid::load_urdf(model_path);
    let mut ethercat_bus = EtherCatMaster::init("eth0", baud_rate: 100_000_000);
    let mut clock = RealTimeClock::new(frequency_hz: 1000);

    let step_height: f64 = 0.18; // 18cm stair step
    let step_length: f64 = 0.28; // 28cm run

    while clock.tick() {
        // 1. Read EtherCAT state (0.10 ms)
        let joint_state = ethercat_bus.read_cyclic_state();
        let imu = ethercat_bus.read_imu_packet();

        // 2. Compute dynamic Capture Point (CP) based on current center of mass velocity
        let com = humanoid.center_of_mass(&joint_state);
        let com_velocity = humanoid.com_velocity(&joint_state);
        let omega = (9.81 / com.z).sqrt();
        let capture_point = [com.x + com_velocity.x / omega, com.y + com_velocity.y / omega];

        // 3. Plan dynamic swing foot trajectory spline over stair edge
        let swing_foot_target = humanoid.calculate_stair_spline(capture_point, step_height, step_length);

        // 4. Solve Whole-Body QP with friction cones (mu = 0.6)
        let wbc_solution = humanoid.solve_wbc_quadprog(
            &joint_state,
            TaskPriority {
                primary: Task::BalanceZMP(capture_point),
                secondary: Task::SwingFootCartesian(swing_foot_target),
                tertiary: Task::PostureRegulate(DefaultPose::Standing),
            },
            FrictionCone::new(coefficient: 0.6)
        );

        // 5. Transmit 34-axis CST torques before 1.00 ms hard deadline
        ethercat_bus.write_cyclic_torques(&wbc_solution.joint_torques);
    }
}

When lifting variable-weight cargo (0 kg to 25 kg) at 2.4 m/s, SERAPHIM dynamically updates link inertia matrices online to prevent tip-over: 

import projects.seraphim.client;
import std.hardware.canfd;

pub fn run_palletizing_loop(can_channel: String) {
    let mut robot = SeraphimHumanoid::load_urdf("models/heavy_worker.urdf");
    let mut bus = CanFdBus::open(can_channel, bitrate: 5_000_000);
    let mut timer = RealTimeClock::new(1000);

    while timer.tick() {
        let state = bus.poll_state();
        
        // 1. Ingest wrist 6-axis load-cell force/torque data
        let wrist_wrench = bus.read_wrist_load_cell();
        
        // 2. Online spatial inertia tensor adjustment for payload link
        let payload_mass = (wrist_wrench.force_z / 9.81).clamp(0.0, 30.0);
        robot.update_end_effector_inertia(
            mass_kg: payload_mass,
            inertia_tensor: Matrix3x3::diagonal([0.05, 0.05, 0.02] * payload_mass)
        );

        // 3. Execute Featherstone RNEA with updated payload parameters
        let inverse_dynamics = robot.compute_rnea(&state, state.desired_acceleration);

        // 4. Dispatch joint torques with feedforward compensation
        bus.write_motor_torques(&inverse_dynamics.torques);
    }
}

Use Case 3: Sub-Millisecond Haptic Teleoperated Surgery

Surgical robotics requires zero GC pause jitter to stream bilateral force feedback at 1 kHz without micro-stutters: 

import projects.seraphim.client;
import std.hardware.ethercat;
import std.network.rt_udp;

pub fn run_teleoperation_surgical_slave() {
    let mut surgical_arm = SeraphimManipulator::load("models/microsurgeon_7dof.urdf");
    let mut eth = EtherCatMaster::init("eth1", 100_000_000);
    let mut udp_socket = RtUdpSocket::bind("192.168.10.50:8000", priority: 99);
    let mut rt_loop = RealTimeClock::new(1000);

    while rt_loop.tick() {
        // Ingest master haptic command packet over low-latency network
        let haptic_packet = udp_socket.receive_fast();
        let arm_state = eth.read_cyclic_state();

        // Compute Cartesian stiffness impedance control
        let cartesian_wrench = surgical_arm.compute_impedance(
            current_pose: arm_state.end_effector_pose,
            target_pose: haptic_packet.desired_pose,
            stiffness: [800.0, 800.0, 800.0], // N/m
            damping: [40.0, 40.0, 40.0]        // N*s/m
        );

        // Convert Cartesian force to joint torques via Jacobian transpose
        let jacobian = surgical_arm.spatial_jacobian(&arm_state.q);
        let joint_torques = jacobian.transpose() * cartesian_wrench;

        // Feedback measured tissue reaction force back to surgeon master console
        let tissue_force = arm_state.tip_force_sensor;
        udp_socket.send_fast(&tissue_force);

        eth.write_cyclic_torques(&joint_torques);
    }
}
