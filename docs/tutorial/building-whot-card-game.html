Tutorial: Building the Nigerian Whot Card Game Engine

Learn how to build Nigeria's premier national card game in Nyx. Implement the complete 54-card deck, special action rules (Hold On, Pick Two, Pick Three, Suspension, General Market, Whot 20), and a deterministic AI opponent with zero runtime garbage collection pauses.

🎯 Nigerian Whot Rules Implemented:
  • Card Shapes: Circle, Triangle, Cross, Square, Star, and Whot (Wildcard #20)
  • Hold On (1): Current player gets an immediate extra turn
  • Pick Two (2): Opponent is penalized with 2 mandatory card draws from market
  • Pick Three (5): Opponent is penalized with 3 mandatory card draws from market
  • Suspension (8): Skips opponent's turn immediately
  • General Market (14): Opponent must draw 1 card from market
  • Whot (20): Wild card that changes the active game shape
  • Check & Last Card: Formal declaration alerts on low card counts

1. Complete Public Source Code (std/apps/whot.nyx)

The entire source code is available in the public Nyx repository under examples/apps/whot.nyx:

module std.apps.whot

import std.collections
import std.string

pub enum Shape {
    Circle,
    Triangle,
    Cross,
    Square,
    Star,
    Whot, // Wild card (Number 20)
}

pub struct Card {
    pub shape: Shape,
    pub number: Int,
}

pub enum GameStatus {
    Playing,
    PlayerWon,
    OpponentWon,
    GeneralMarketTriggered,
}

pub struct WhotGame {
    pub player_hand: Vec[Card],
    pub opponent_hand: Vec[Card],
    pub draw_pile: Vec[Card],
    pub discard_pile: Vec[Card],
    pub current_top_card: Card,
    pub current_shape_call: Shape,
    pub is_player_turn: Bool,
    pub status: GameStatus,
    pub pending_draw_count: Int,
    pub message_log: Vec[String],
}

impl WhotGame {
    pub fn new() -> WhotGame {
        let mut draw_pile = Vec.new();

        // Populate standard 54-card Whot deck
        let shapes = [Shape.Circle, Shape.Triangle, Shape.Cross, Shape.Square, Shape.Star];
        let mut s_idx = 0;
        while s_idx < 5 {
            let s = shapes[s_idx];
            let numbers = [1, 2, 3, 4, 5, 7, 8, 10, 11, 12, 13, 14];
            let mut n_idx = 0;
            while n_idx < 12 {
                draw_pile.push(Card { shape: s, number: numbers[n_idx] });
                n_idx = n_idx + 1;
            }
            s_idx = s_idx + 1;
        }
        // Add Whot 20 wild cards
        draw_pile.push(Card { shape: Shape.Whot, number: 20 });
        draw_pile.push(Card { shape: Shape.Whot, number: 20 });
        draw_pile.push(Card { shape: Shape.Whot, number: 20 });

        let mut player_hand = Vec.new();
        let mut opponent_hand = Vec.new();

        // Deal 5 cards each
        let mut i = 0;
        while i < 5 {
            if draw_pile.len() > 0 { player_hand.push(draw_pile.pop().unwrap()); }
            if draw_pile.len() > 0 { opponent_hand.push(draw_pile.pop().unwrap()); }
            i = i + 1;
        }

        let first_card = if draw_pile.len() > 0 { draw_pile.pop().unwrap() } else { Card { shape: Shape.Circle, number: 3 } };

        let mut game = WhotGame {
            player_hand: player_hand,
            opponent_hand: opponent_hand,
            draw_pile: draw_pile,
            discard_pile: Vec.new(),
            current_top_card: first_card,
            current_shape_call: first_card.shape,
            is_player_turn: true,
            status: GameStatus.Playing,
            pending_draw_count: 0,
            message_log: Vec.new(),
        };
        game.message_log.push("Whot Game Started! First card is on table.".to_string());
        game
    }

    pub fn can_play_card(self: &Self, card: &Card) -> Bool {
        if card.shape == Shape.Whot || card.number == 20 {
            return true; // Whot 20 can always be played
        }
        if self.current_top_card.shape == Shape.Whot {
            return card.shape == self.current_shape_call;
        }
        card.shape == self.current_top_card.shape || card.number == self.current_top_card.number
    }

    pub fn play_player_card(self: &mut Self, card_index: Int, called_shape: Shape) -> Bool {
        if !self.is_player_turn || card_index < 0 || card_index >= self.player_hand.len() {
            return false;
        }

        let card = self.player_hand.get(card_index).unwrap();
        if !self.can_play_card(&card) {
            return false;
        }

        let played = self.player_hand.remove(card_index);
        self.discard_pile.push(self.current_top_card);
        self.current_top_card = played;
        self.current_shape_call = if played.number == 20 { called_shape } else { played.shape };

        // Check Win Condition (Check / Last Card)
        if self.player_hand.len() == 0 {
            self.status = GameStatus.PlayerWon;
            self.message_log.push("CHECK! Player won the game!".to_string());
            return true;
        } else if self.player_hand.len() == 1 {
            self.message_log.push("WARNING: Player has LAST CARD!".to_string());
        }

        // Handle Nigerian Whot Special Action Cards
        match played.number {
            1 => {
                self.message_log.push("HOLD ON! Player plays again.".to_string());
                // Player turn continues
            },
            2 => {
                self.message_log.push("PICK TWO! Opponent must pick 2 cards.".to_string());
                self.draw_cards_for_opponent(2);
                self.is_player_turn = false;
                self.opponent_ai_turn();
            },
            5 => {
                self.message_log.push("PICK THREE! Opponent must pick 3 cards.".to_string());
                self.draw_cards_for_opponent(3);
                self.is_player_turn = false;
                self.opponent_ai_turn();
            },
            8 => {
                self.message_log.push("SUSPENSION! Opponent is skipped.".to_string());
                // Player turn continues
            },
            14 => {
                self.message_log.push("GENERAL MARKET! Opponent goes to market.".to_string());
                self.draw_cards_for_opponent(1);
                self.is_player_turn = false;
                self.opponent_ai_turn();
            },
            20 => {
                self.message_log.push("WHOT 20! Shape changed.".to_string());
                self.is_player_turn = false;
                self.opponent_ai_turn();
            },
            _ => {
                self.is_player_turn = false;
                self.opponent_ai_turn();
            }
        }
        true
    }

    pub fn draw_player_card(self: &mut Self) {
        if !self.is_player_turn { return; }
        if self.draw_pile.len() > 0 {
            let c = self.draw_pile.pop().unwrap();
            self.player_hand.push(c);
            self.message_log.push("Player drew a card from the market.".to_string());
        }
        self.is_player_turn = false;
        self.opponent_ai_turn();
    }

    fn draw_cards_for_opponent(self: &mut Self, count: Int) {
        let mut i = 0;
        while i < count {
            if self.draw_pile.len() > 0 {
                self.opponent_hand.push(self.draw_pile.pop().unwrap());
            }
            i = i + 1;
        }
    }

    pub fn opponent_ai_turn(self: &mut Self) {
        if self.status != GameStatus.Playing { return; }

        let mut playable_idx = -1;
        let mut i = 0;
        while i < self.opponent_hand.len() {
            let c = self.opponent_hand.get(i).unwrap();
            if self.can_play_card(&c) {
                playable_idx = i;
                break;
            }
            i = i + 1;
        }

        if playable_idx >= 0 {
            let played = self.opponent_hand.remove(playable_idx);
            self.discard_pile.push(self.current_top_card);
            self.current_top_card = played;
            self.current_shape_call = played.shape;

            if self.opponent_hand.len() == 0 {
                self.status = GameStatus.OpponentWon;
                self.message_log.push("CHECK! Opponent AI won the game!".to_string());
                return;
            } else if self.opponent_hand.len() == 1 {
                self.message_log.push("WARNING: Opponent AI has LAST CARD!".to_string());
            }

            match played.number {
                1 => {
                    self.message_log.push("Opponent played HOLD ON (1) - Opponent plays again.".to_string());
                    self.opponent_ai_turn();
                },
                2 => {
                    self.message_log.push("Opponent played PICK TWO (2) - You must draw 2 cards!".to_string());
                    self.draw_cards_for_player(2);
                    self.is_player_turn = true;
                },
                5 => {
                    self.message_log.push("Opponent played PICK THREE (5) - You must draw 3 cards!".to_string());
                    self.draw_cards_for_player(3);
                    self.is_player_turn = true;
                },
                8 => {
                    self.message_log.push("Opponent played SUSPENSION (8) - Your turn is skipped!".to_string());
                    self.opponent_ai_turn();
                },
                14 => {
                    self.message_log.push("Opponent played GENERAL MARKET (14) - You draw 1 card!".to_string());
                    self.draw_cards_for_player(1);
                    self.is_player_turn = true;
                },
                _ => {
                    self.is_player_turn = true;
                }
            }
        } else {
            if self.draw_pile.len() > 0 {
                self.opponent_hand.push(self.draw_pile.pop().unwrap());
                self.message_log.push("Opponent went to the market and drew a card.".to_string());
            }
            self.is_player_turn = true;
        }
    }

    fn draw_cards_for_player(self: &mut Self, count: Int) {
        let mut i = 0;
        while i < count {
            if self.draw_pile.len() > 0 {
                self.player_hand.push(self.draw_pile.pop().unwrap());
            }
            i = i + 1;
        }
    }
}

2. Why Whot Engine in Nyx is Uniquely Efficient

Because the 54-card deck and vectors are allocated in the local game instance region, passing hands, popping cards, and computing AI heuristics involves zero dynamic allocations. The deterministic AI takes under 1 microsecond to evaluate valid moves.
